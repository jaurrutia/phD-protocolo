\begin{frame}{Minimización del error mediante descenso de gradiente}
                {Error por mínimos cuadrados}
    \begin{itemize}
            \item Descenso de gradiente es un método iterativo para obtener el mínimo de una función $F$:
         $$ \vec{x}_{n+1} = \vec{x}_n - \gamma \nabla F(\vec{x}_n), $$
         donde $\vec{x}_{n+1}$ es un punto tal que $F(\vec{x}_n)\sim F_\text{min}$, y $\gamma < 1$ es conocido como \textit{learning rate}.
         \item Para realizar el mejor ajuste se minimiza la función de error dada por el método de mínimos cuadrados, es decir:
         $$  \text{error}(\vec{x}; \{f_i\}_{i\leq n},\{y_i\}_{i\leq n}) = \frac{1}{n}\sum_{i = 1}^n \qty(f_i(\vec{x}) - y_i)^2, $$
         con $f_i(\vec{x})$ el modelo teórico, $y_i$ la lista de valores a ajustar el modelo y $n$ el número de datos.
         \item  Tolerancia empleada en los cálculos: $$\norm{\vec{x}_{n+1} - \vec{x}_n}<10^{-8}.$$
         \item Para el \textit{learning rate} se empleó el método de Barzilai–Borwein$^1$ que da una expresión variable de $\gamma = \gamma_n$, que garantiza convergencia a un mínimo local (en caso de existir) y cuya expresión es:
         $$ \gamma_n = \frac{\qty(\vec{x}_{n} - \vec{x}_{n-1}) \cdot  \qty[\nabla F(\vec{x}_n)-\nabla F(\vec{x}_{n-1})]}
                            {\norm{\nabla F(\vec{x}_n)-\nabla F(\vec{x}_{n-1})}^2}. $$
    \end{itemize}

    \vspace*{1em}
    \noindent\rule{.25\textwidth}{0.4pt}\\
\fontsize{4}{5} \selectfont
	$^1$ \fullcite{barzilai1988two-point}
\end{frame}

    \end{itemize}
