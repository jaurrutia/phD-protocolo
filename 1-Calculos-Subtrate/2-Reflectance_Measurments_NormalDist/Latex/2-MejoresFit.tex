
\frame{
    \vfill
		\begin{beamercolorbox}[sep=6pt]{part title}
		\usebeamerfont{section title}
		Mejores ajustes (Corrección en la normalización de las reflectancias):
    	\end{beamercolorbox}
	\begin{beamercolorbox}[sep=8pt]{part title}
	\begin{columns}
	\column{.5\textwidth}
     	\begin{itemize}
     	\item Parámetros de ajuste
			\begin{itemize}
			\item Radio $a$ y fracción de cubierta $\Theta$\\[2em]
			\end{itemize}
     	\item Ventana espectral de ajuste:
			\begin{itemize}
				\item Polarización $p$: $450\text{ nm}\leq \lambda \leq 700\text{ nm}$
				\item Polarización $s$: $450\text{ nm}\leq \lambda \leq 850\text{ nm}$
			\end{itemize}
     	\end{itemize}
     \column{.5\textwidth}
        \begin{itemize}
        \item Los valores de radio que mejor ajustan son $\sim 2$ nm menores al valor nominal\\[1em]
        \item Polarización $p$
            \begin{itemize}
            \item Fracciones de cubierta alrededor del $15\%$ y $20\%$ para las mediciones a $64.9^\circ$ y $72^\circ$, respectivamente
            \item Buen ajuste en mangnitud en todo el visible
            \item Corrimiento al rojo de la LSPR respecto al resultado experimental\\[1em]
            \end{itemize}
        \item Polarización $s$
            \begin{itemize}
            \item Fracciones de cubierta alrededor del $38\%$ para todas las mediciones
            \item Buen ajuste en mangnitud en todo el visible
            \item Corrimiento al azul de la LSPR respecto al resultado experimental
            \end{itemize}
        \item De forma general, los ajustes muestran un mejor acuerdo con el experimento para $\theta_i = 72.9^\circ$
        \end{itemize}

    \end{columns}
	    \end{beamercolorbox}
    \vfill
}
